# The 1.26.4 pin arrived with the first commit carrying no reason. Checked
# rather than assumed before moving it: extracting six remotes under 1.26.4
# and under 2.2.6, same OpenCV, gave **byte-identical fingerprints and
# byte-identical crops**, verification is ~11% faster (3.27 ms -> 2.91 ms per
# candidate at 45 buttons), and retrieval is unchanged at recall@1 8/8,
# separation +0.077.
#
# It only became installable once the OCR wrappers moved to
# requirements-ocr.txt: rapidocr-onnxruntime declares `numpy<2.0.0`, and pip
# resolves the metadata of anything listed in a requirements file even when
# something else installs it --no-deps later.
numpy==2.2.6
scipy==1.13.1
# THE OPENCV VERSION CHANGES EXTRACTION. Do not let it drift, and never let
# the full `opencv-python` build be installed alongside this one -- it shadows
# the headless build and silently swaps the CV library out from under the
# pipeline. Measured on the same six remotes, same numpy, 5.0.0.93 against
# 4.10.0.84: **every fingerprint differed**, six of seven by button count
# (RM-L859-1 47 vs 50, DVD_80 73 vs 75, GINZZU_GM-501_1 lost its only text
# region). That is the query/catalog asymmetry this project keeps finding,
# arriving through the dependency tree instead of through the code.
opencv-python-headless==4.10.0.84
Pillow==10.4.0

# OCR. RapidOCR runs the PP-OCR detection/recognition models under
# onnxruntime: the same model family as PaddleOCR at ~150 MB rather than
# ~1 GB, which is what fits this box. The engine is swappable behind
# app/pipeline/ocr.py -- see OCR_EXTRA in install.sh to add paddle/easyocr/
# doctr/tesseract for engine consensus on a larger machine.
#
# The wrappers themselves are NOT here -- see requirements-ocr.txt, which is
# installed --no-deps. This file holds only what they really need at runtime,
# which is pinned and resolvable.
onnxruntime==1.23.2

# The same PP-OCR models under OpenVINO. Installed alongside rather than
# instead: it is roughly twice as fast on identical models (see
# RapidOcrOpenvinoEngine for the figures) but it reads *more* text, so it
# changes fingerprints and cannot be adopted on one path only. Having both in
# the image is what allows the two to be A/B'd on the box that matters, which
# is the only place the choice can honestly be made. Selected with
# RCU_OCR_ENGINE=openvino; the default is unchanged.
# The runtime, not the wrapper: rapidocr-openvino is in requirements-ocr.txt.
openvino==2024.0.0
pyclipper==1.3.0.post6
Shapely==2.0.5
PyYAML==6.0.2
six==1.16.0
